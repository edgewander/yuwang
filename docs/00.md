(function() {
  // 从sidebar读取文章列表
  fetch('_sidebar.md')
    .then(r => r.text())
    .then(text => {
      var lines = text.split('\n');
      var articles = [];
      lines.forEach(function(line) {
        var m = line.match(/\*\s*\[(.+?)\]\((.+?)\)/);
        if (m) articles.push({ title: m[1], file: m[2] });
      });

      var cats = {
        art:      { el: 'cat-art',      test: f => /^3\./.test(f) },
        photo:    { el: 'cat-photo',    test: f => /^9\./.test(f) },
        painting: { el: 'cat-painting', test: f => /^8\./.test(f) },
        poetry:   { el: 'cat-poetry',   test: f => /^7\./.test(f) },
        history:  { el: 'cat-history',  test: f => /^2\./.test(f) },
        music:    { el: 'cat-music',    test: f => /^6\./.test(f) },
        travel:   { el: 'cat-travel',   test: f => /^11\./.test(f) },
        misc:     { el: 'cat-misc',     test: f => /^(1\.|5\.|10\.)/.test(f) },
      };

      Object.values(cats).forEach(function(cat) {
        var items = articles.filter(a => cat.test(a.file));
        var html = items.map(function(a) {
          var id = a.file.replace('.md', '');
          return '<div style="border-top:1px solid #1e1b18;padding:8px 0;">' +
            '<a href="#/' + id + '" style="color:#999;font-size:13px;text-decoration:none;border:none;line-height:1.5;display:block;">' +
            a.title + '</a></div>';
        }).join('');
        var el = document.getElementById(cat.el);
        if (el) el.innerHTML = html || '<div style="color:#555;font-size:12px;padding:8px 0;">暂无文章</div>';
      });
    })
    .catch(function() {
      console.log('sidebar load failed');
    });
})();
</script>
